UPDATE SHOPS s
LEFT JOIN SHOP_FORMATS_STACK_RANGE f
  ON ROUND(
       (COALESCE(s.LONG_STACK_COUNT, 0) / 1.5) - COALESCE(s.LONG_STACK_COUNT, 0) 
       + COALESCE(s.STACK_COUNT, 0) - COALESCE(s.SHORT_STACK_COUNT, 0)
     ) BETWEEN f.STACK_FROM AND f.STACK_TO
SET s.STACK_RECALC_COUNT = ROUND(
       (COALESCE(s.LONG_STACK_COUNT, 0) / 1.5) - COALESCE(s.LONG_STACK_COUNT, 0) 
       + COALESCE(s.STACK_COUNT, 0) - COALESCE(s.SHORT_STACK_COUNT, 0)
     ),
    s.STORE_CLASS = COALESCE(f.FORMAT, 'Диапозон количества стеллажей "Формат MX" задан не верно');
